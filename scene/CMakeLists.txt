cmake_minimum_required(VERSION 3.16)
project(mmo_scene LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(scene-core src/scene.cpp)
target_include_directories(scene-core PUBLIC src)
target_include_directories(scene-core SYSTEM PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../third_party")
if(MSVC)
  target_compile_options(scene-core PUBLIC /utf-8 /W4)
else()
  target_compile_options(scene-core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(scene-server src/main.cpp)
target_link_libraries(scene-server PRIVATE scene-core Threads::Threads)
target_compile_definitions(scene-server PRIVATE CPPHTTPLIB_THREAD_POOL_COUNT=8)
if(WIN32)
  target_compile_definitions(scene-server PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(scene-server PRIVATE ws2_32)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(scene-tests tests/scene_tests.cpp)
  target_link_libraries(scene-tests PRIVATE scene-core)
  add_test(NAME scene-state-invariants COMMAND scene-tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME scene-http-lifecycle COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/http_tests.py" "$<TARGET_FILE:scene-server>")
  endif()
endif()
