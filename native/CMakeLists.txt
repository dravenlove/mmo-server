cmake_minimum_required(VERSION 3.16)
project(mmo_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MARIADB REQUIRED IMPORTED_TARGET libmariadb)
find_package(Boost 1.74 REQUIRED COMPONENTS system)

add_library(native-common INTERFACE)
target_include_directories(native-common INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(native-common SYSTEM INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/../third_party")
target_link_libraries(native-common INTERFACE Threads::Threads OpenSSL::Crypto OpenSSL::SSL)
target_compile_definitions(native-common INTERFACE CPPHTTPLIB_THREAD_POOL_COUNT=8 CPPHTTPLIB_OPENSSL_SUPPORT)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(native-common INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_subdirectory(persistence)
add_subdirectory(account)
add_subdirectory(game_service)
add_subdirectory(gateway)
add_subdirectory(script)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../scene" "${CMAKE_CURRENT_BINARY_DIR}/scene-core")
add_subdirectory(room)
