find_library(NATIVE_CRYPT_LIBRARY NAMES crypt REQUIRED)
add_library(native-account-core account.cpp)
target_link_libraries(native-account-core PUBLIC native-common native-db ${NATIVE_CRYPT_LIBRARY})
target_include_directories(native-account-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_executable(native-account main.cpp)
target_link_libraries(native-account PRIVATE native-account-core)
target_compile_definitions(native-account PRIVATE CPPHTTPLIB_HEADER_MAX_LENGTH=16384)
if(BUILD_TESTING)
  add_executable(native-account-unit tests/account_tests.cpp)
  target_link_libraries(native-account-unit PRIVATE native-account-core)
  add_test(NAME native-account-invariants COMMAND native-account-unit)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME native-account-http COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/http_tests.py" "$<TARGET_FILE:native-account>")
  set_tests_properties(native-account-http PROPERTIES TIMEOUT 60)
endif()
