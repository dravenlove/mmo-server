find_package(Boost 1.74 REQUIRED COMPONENTS system)
find_package(Threads REQUIRED)
add_library(native-gateway-core protocol.cpp)
target_include_directories(native-gateway-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party")
target_compile_features(native-gateway-core PUBLIC cxx_std_17)
target_link_libraries(native-gateway-core PUBLIC Threads::Threads)
add_executable(native-gateway main.cpp runtime.cpp)
target_link_libraries(native-gateway PRIVATE native-gateway-core native-common Boost::system Threads::Threads)
if(MSVC)
  target_compile_options(native-gateway PRIVATE /utf-8 /W4)
  target_compile_options(native-gateway-core PRIVATE /utf-8 /W4)
else()
  target_compile_options(native-gateway PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(native-gateway-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(native-gateway-protocol-tests tests/protocol_tests.cpp)
  target_link_libraries(native-gateway-protocol-tests PRIVATE native-gateway-core)
  add_test(NAME native-gateway-protocol COMMAND native-gateway-protocol-tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME native-gateway-http-integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:native-gateway>")
  endif()
endif()
